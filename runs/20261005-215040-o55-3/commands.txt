def gitrepo(path):
    subprocess.run(["git", "init", "-q", path], check=True)

cases = {}
def case(f): cases[f.__name__] = f; return f

@case
def symlink_gitignore(r):
    W(r, "real", b"*.log\n"); os.symlink("real", os.path.join(r, ".gitignore")); W(r, "a.log")

def negate_into_excluded_dir(r):
