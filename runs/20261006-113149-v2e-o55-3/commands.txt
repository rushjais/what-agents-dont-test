$ mkdir -p /tmp/fuzz && cat > /tmp/fuzz/fuzz.py <<'EOF'
import os, random, shutil, subprocess, sys, tempfile
sys.path.insert(0, "/work/repo")
from snapshot.ignore import ignored_files, PackignoreError

NAMES = ["a", "b", "A", "ab", "a.tmp", "b.tmp", "x", "d", "D", "s", "t", ".packignore", "build",
         "ß", "ss", "é", "é", "[a]", "{a,b}", "a b", "!a", ";c", "#c", "\\x", "*", "?"]
CHARS = ["a", "b", "x", "d", "s", "t", "*", "**", "?", "/", "{", "}", ",", "[", "]", "!", " ",
         ";", "#", "\\", ".", "tmp", "A", "-", "\t", "\r"]

def rand_pattern(r):
    p = "".join(r.choice(CHARS) for _ in range(r.randint(1, 7)))
    if r.random() < 0.3: p = "!" + p
    return p

def rand_tree(r):
    files = {}
    dirs = [""]
    for _ in range(r.randint(1, 12)):
        d = r.choice(dirs)
        name = r.choice(NAMES)
        path = (d + "/" + name) if d else name
        if r.random() < 0.3 and name != ".packignore":
            dirs.append(path)
        else:
            files[path] = None
    for d in r.sample(dirs, min(len(dirs), r.randint(1, 3))):
        pi = (d + "/.packignore") if d else ".packignore"
        sep = r.choice(["\n", "\r\n"])
        files[pi] = sep.join(rand_pattern(r) for _ in range(r.randint(1, 6))) + sep
    return files

def make(files, root):
    for p, c in files.items():
        full = os.path.join(root, p)
        try:
            os.makedirs(os.path.dirname(full), exist_ok=True)
            if os.path.isdir(full): return False
            with open(full, "w", newline="") as f: f.write(c or "")
        except (NotADirectoryError, FileExistsError, IsADirectoryError):
            return False
    return True

def run(seed):
    r = random.Random(seed)
    files = rand_tree(r)
    root = tempfile.mkdtemp(prefix="fz")
    try:
        if not make(files, root): return None
        ref = subprocess.run([sys.executable, "/work/repo/tools/packignore", root], capture_output=True, text=True)
        try:
            mine = ignored_files(root); merr = None
        except PackignoreError as e:
            mine = None; merr = str(e)
        if ref.returncode != 0:
            if merr is None: return ("ref error, mine ok", ref.stderr, mine)
            return None
        exp = sorted(l for l in ref.stdout.splitlines() if l)
        if merr: return ("mine error", merr, exp)
        if exp != mine:
            return ("diff", sorted(set(exp) - set(mine)), sorted(set(mine) - set(exp)))
        return None
    finally:
        shutil.rmtree(root)

if __name__ == "__main__":
    start, n = int(sys.argv[1]), int(sys.argv[2])
    bad = 0
    for s in range(start, start + n):
        res = run(s)
        if res:
            bad += 1
            r = random.Random(s); files = rand_tree(r)
            print("SEED", s, res)
            for k, v in files.items():
                if v is not None: print("   ", repr(k), repr(v))
            print("    files:", [k for k, v in files.items() if v is None])
            if bad > 5: break
    print("bad", bad)
EOF
