import random, os, sys, tempfile, shutil, subprocess, json
sys.path.insert(0, '/work/repo')
from snapshot.ignore import ignored_files

NAMES = ['a','b','d','e','ab','ba','a.b','.a','x y','#a','!a',';a','[a]','a\\','A','é','*','?']
SEGS = ['a','b','d','e','*','?','**','a*','*a','?b','a?','.a','*.b','a.b','ab','#a','!a','[a]','A','\\a','***','x y','']
def rand_name(r): return r.choice(NAMES[:8] if r.random()<0.7 else NAMES)
def rand_pattern(r):
    n = r.choice([1,1,1,2,2,3,4])
    p = '/'.join(r.choice(SEGS[:12] if r.random()<0.7 else SEGS) for _ in range(n))
    if r.random()<0.25: p = '/' + p
    if r.random()<0.25: p = p + '/'
    if r.random()<0.05: p = p + '//'
    if r.random()<0.05: p = '//' + p
    if r.random()<0.35: p = '!' + p
    if r.random()<0.05: p = r.choice([';',' ;','#',' ','\t','!!','! ']) + p
    if r.random()<0.05: p = p + r.choice([' ','\t','\r','\x0b'])
    if r.random()<0.03: p = ''
    return p
def rand_tree(r):
    dirs = ['']
    files = {}
    for _ in range(r.randint(1, 6)):
        parent = r.choice(dirs)
        d = (parent + '/' if parent else '') + rand_name(r)
        dirs.append(d)
    for _ in range(r.randint(1, 14)):
        d = r.choice(dirs)
        files[(d + '/' if d else '') + rand_name(r)] = ''
    for d in r.sample(dirs, min(len(dirs), r.randint(1, 3))):
        lines = [rand_pattern(r) for _ in range(r.randint(1, 6))]
        files[(d + '/' if d else '') + '.packignore'] = '\n'.join(lines) + r.choice(['','\n'])
    # remove conflicts (file vs dir)
    paths = sorted(files)
    out = {}
    for p in paths:
        if any(o.startswith(p + '/') for o in paths): continue
        out[p] = files[p]
    for p in out:
        if r.random()<0.06: out[p] = r.choice(['@generated\n', 'x\n@generated', 'foo @generated'])
    modes = {p: 0o755 for p in out if r.random()<0.05}
    return out, modes

def build(files, modes):
    d = tempfile.mkdtemp(dir='/tmp/pi/trees')
    for p, c in files.items():
        full = os.path.join(d, p)
        os.makedirs(os.path.dirname(full), exist_ok=True)
        with open(full, 'w') as f: f.write(c)
        if p in modes: os.chmod(full, modes[p])
    return d

def oracle(d):
    r = subprocess.run([sys.executable, '/work/repo/tools/packignore', d], capture_output=True, text=True)
    if r.returncode: return 'ERR ' + r.stderr
    return sorted(l for l in r.stdout.splitlines() if l)

if __name__ == '__main__':
    seed0 = int(sys.argv[1]); n = int(sys.argv[2])
    bad = 0
    for s in range(seed0, seed0+n):
        r = random.Random(s)
        files, modes = rand_tree(r)
        d = build(files, modes)
        try:
            exp = oracle(d)
            try: got = ignored_files(d)
            except Exception as e: got = 'EXC ' + repr(e)
            if exp != got:
                bad += 1
                print('SEED', s)
                for p in sorted(files):
                    if p.endswith('.packignore'): print('  PI', repr(p), repr(files[p]))
                print('  files', sorted(p for p in files if not p.endswith('.packignore')), 'modes', modes)
                print('  exp-got', sorted(set(exp)-set(got)) if isinstance(exp,list) and isinstance(got,list) else exp)
                print('  got-exp', sorted(set(got)-set(exp)) if isinstance(exp,list) and isinstance(got,list) else got)
        finally:
            shutil.rmtree(d)
    print('bad', bad, 'of', n)

            if os.path.islink(full) or not os.path.isfile(full):
                continue
            parts = os.path.relpath(full, root).split(os.sep)
            key = ""
            spelled = []
            for i, part in enumerate(parts):
                key += "/" + _fold(part)
                kind = "file" if i == len(parts) - 1 else "dir"
                if kinds.setdefault(key, kind) != kind:
                    raise ValueError(f"packignore: {'/'.join(parts)} collides with a "
                                     f"{kinds[key]} of the same name")
                spelled.append(names.setdefault(key, part))
            sources["/".join(spelled)] = full
    return sources
