$ cat > /tmp/pq.py << 'PYEOF'
"""Query packignore service directly."""
import json, os, sys, urllib.request
URL = os.environ["PACKIGNORE_URL"]
# files is a dict rel-path -> content-or-None
def query(files):
    req = urllib.request.Request(URL + "/ignored", data=json.dumps({"files": files}).encode(),
                                 headers={"Content-Type": "application/json"})
    with urllib.request.urlopen(req, timeout=60) as resp:
        return json.load(resp)["ignored"]
if __name__ == "__main__":
    data = json.load(sys.stdin)
    for p in query(data):
        print(p)
PYEOF
echo "ok"
