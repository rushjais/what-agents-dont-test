from t import run
import sys
def cmp(P,Q,files,target):
    r1 = target in run(files,{"":P+"\n!"+Q})
    r2 = target in run(files,{"":"!"+Q+"\n"+P})
    m1 = target in run(files,{"":P}); m2 = target in run(files,{"":"*\n!"+Q})
    tag = {(True,True):'P>Q',(False,False):'P<Q',(False,True):'EQ',(True,False):'EARLIER?'}[(r1,r2)]
    return tag + ('' if m1 and not m2 else f' [match P={m1} Q={not m2}]')
if __name__=='__main__':
    target=sys.argv[1]; files=[target]+sys.argv[2:]
    for line in sys.stdin:
        line=line.rstrip('\n')
        if not line: continue
        P,Q=line.split(' || ')
        print(f'{P!r:20} {Q!r:20}', cmp(P,Q,files,target))
