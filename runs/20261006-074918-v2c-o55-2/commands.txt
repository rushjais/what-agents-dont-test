import json, sys, urllib.request, os
URL = os.environ.get("PACKIGNORE_URL", "http://127.0.0.1:8765")
def q(files):
    req = urllib.request.Request(URL + "/ignored", data=json.dumps({"files": files}).encode(),
                                 headers={"Content-Type": "application/json"})
    try:
        with urllib.request.urlopen(req, timeout=60) as r:
            return json.load(r)["ignored"]
    except urllib.error.HTTPError as e:
        return "ERR:" + str(json.load(e))
def t(pi, paths, extra=None):
    files = {".packignore": pi}
    for p in paths: files[p] = None
    if extra: files.update(extra)
    return q(files)
EOF
