import os, sys, random, shutil, subprocess, tempfile, json
sys.path.insert(0, "/work/repo")
from snapshot.ignore import ignored_files
NAMES = ["a","b","c","ab","ba","x.t","y.t",".h","a.b","[a]","!a","#a","a b","*","?","\\a","é","aa","a*",";a","**"]
DIRN = ["a","b","c","ab","d","x.t","[a]",".h","**","!a"]
PTOK = ["a","b","c","ab","x","t",".","*","*","**","?","/","/","/","[a]","!","#","\\"," ","é","h",".t",";"]
def gen_pat(rng, depthnames):
    r = rng.random()
    if r < 0.3:
        p = "".join(rng.choice(PTOK) for _ in range(rng.randint(1, 6)))
    else:
        segs = []
        for _ in range(rng.randint(1, 4)):
            k = rng.random()
            if k < 0.2: segs.append("**")
            elif k < 0.35: segs.append("*")
            elif k < 0.45: segs.append(rng.choice(["a*","*.t","?","a?","*b","a**","**b"]))
            else: segs.append(rng.choice(depthnames))
        p = "/".join(segs)
        if rng.random() < 0.3: p = "/" + p
        if rng.random() < 0.3: p = p + "/"
    if rng.random() < 0.3: p = "!" + p
    if rng.random() < 0.1: p = rng.choice([" ", "\t", "  "]) + p
    if rng.random() < 0.1: p = p + rng.choice([" ", "\t", "\r"])
    if rng.random() < 0.05: p = rng.choice([";", "#", "!!", "\\!", "\\#"]) + p
    return p
def gen_tree(rng):
    files = {}
    dirs = [""]
    for _ in range(rng.randint(1, 8)):
        par = rng.choice(dirs)
        d = (par + "/" if par else "") + rng.choice(DIRN)
        if d not in dirs: dirs.append(d)
    for _ in range(rng.randint(1, 15)):
        par = rng.choice(dirs)
        f = (par + "/" if par else "") + rng.choice(NAMES)
        files[f] = ""
    allnames = DIRN + NAMES
    for d in dirs:
        if rng.random() < (0.9 if d == "" else 0.35):
            n = rng.randint(1, 6)
            files[(d + "/" if d else "") + ".packignore"] = "\n".join(gen_pat(rng, allnames) for _ in range(n)) + "\n"
    # drop files that collide with dirs
    paths = set(files)
    bad = {p for p in paths for q in paths if q.startswith(p + "/")}
    for d in dirs:
        bad |= {p for p in paths if p == d}
    for p in bad: files.pop(p, None)
    # case collisions
    seen = {}
    for p in list(files):
        k = p.lower()
        if k in seen: files.pop(p)
        else: seen[k] = p
    return files
def write(base, files):
    for p, c in files.items():
        fp = os.path.join(base, p)
        os.makedirs(os.path.dirname(fp), exist_ok=True)
        with open(fp, "w") as f: f.write(c)
def run(seed, batch=40):
    rng = random.Random(seed)
    top = tempfile.mkdtemp(dir="/tmp/pk/fz")
    cases = []
    for i in range(batch):
        t = gen_tree(rng)
        # avoid lowercase collisions with case dir names
        write(os.path.join(top, "c%d" % i), t)
        cases.append(t)
    # oracle on each case individually is slow; use root per case via subdir trick:
    r = subprocess.run([sys.executable, "/work/repo/tools/packignore", top], capture_output=True, text=True)
    if r.returncode:
        print("ERR", seed, r.stderr); return top, None
    got = [l for l in r.stdout.splitlines() if l]
    mine = ignored_files(top)
    if got != mine:
        for i in range(batch):
            pre = "c%d/" % i
            g = [x for x in got if x.startswith(pre)]; m = [x for x in mine if x.startswith(pre)]
            if g != m:
                print("MISMATCH seed", seed, "case", i)
                print(json.dumps(cases[i], indent=1, ensure_ascii=False))
                print("oracle only:", sorted(set(g) - set(m)))
                print("mine only:", sorted(set(m) - set(g)))
                return top, False
    shutil.rmtree(top)
    return None, True
if __name__ == "__main__":
    s0 = int(sys.argv[1]); n = int(sys.argv[2])
    for s in range(s0, s0 + n):
        top, ok = run(s)
        if not ok: break
    else:
        print("all ok", s0, n)
EOF
