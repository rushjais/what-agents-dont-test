import os, sys, json, subprocess, tempfile, shutil
def run(files):
    d = tempfile.mkdtemp(dir='/tmp/w')
    for p, c in files.items():
        full = os.path.join(d, p)
        os.makedirs(os.path.dirname(full), exist_ok=True)
        open(full, 'w').write(c)
    r = subprocess.run([sys.executable, '/work/repo/tools/packignore', d], capture_output=True, text=True)
    shutil.rmtree(d)
    return r.stdout.split() if r.returncode == 0 else ('ERR', r.stderr.strip())
if __name__ == '__main__':
    files = json.loads(sys.argv[1])
    print(run(files))
EOF
