import io, json, tarfile, urllib.request, urllib.error
URL="http://127.0.0.1:8765"
def send(entries):
    """entries: list of (name, content, mode=0o644)"""
    buf=io.BytesIO()
    with tarfile.open(fileobj=buf, mode="w:gz") as tar:
        for e in entries:
            name, c = e[0], e[1]
            mode = e[2] if len(e)>2 else 0o644
            c = c.encode() if isinstance(c,str) else c
            ti=tarfile.TarInfo(name); ti.size=len(c); ti.mode=mode; ti.type=tarfile.REGTYPE
            tar.addfile(ti, io.BytesIO(c))
    req=urllib.request.Request(URL+"/ignored", data=buf.getvalue(), headers={"Content-Type":"application/x-tar"})
    try:
        with urllib.request.urlopen(req, timeout=60) as r: return json.load(r)["ignored"]
    except urllib.error.HTTPError as e:
        return "ERR:"+e.read().decode()
