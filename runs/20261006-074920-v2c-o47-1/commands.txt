$ mkdir -p /tmp/probe && cat > /tmp/probe/probe.py <<'EOF'
"""Probe the packignore service directly. Takes a dict {path: content or None}."""
import json, urllib.request, os, sys
URL = os.environ["PACKIGNORE_URL"]

def probe(files):
    req = urllib.request.Request(URL + "/ignored", data=json.dumps({"files": files}).encode(), headers={"Content-Type":"application/json"})
    with urllib.request.urlopen(req) as r:
        return json.load(r)["ignored"]

if __name__ == "__main__":
    files = json.load(sys.stdin)
    print(json.dumps(probe(files), indent=2))
EOF

    req = urllib.request.Request(URL + "/ignored",
