URL = os.environ["PACKIGNORE_URL"]
def q(files):
    """files: dict path -> content(str/bytes) or (content, mode)"""
    buf = io.BytesIO()
    with tarfile.open(fileobj=buf, mode="w:gz") as tar:
        for p, v in files.items():
            mode = 0o644
            if isinstance(v, tuple): v, mode = v
            if isinstance(v, str): v = v.encode()
            ti = tarfile.TarInfo(p); ti.size = len(v); ti.mode = mode; ti.mtime = time.time()
            tar.addfile(ti, io.BytesIO(v))
    req = urllib.request.Request(URL + "/ignored", data=buf.getvalue(), headers={"Content-Type": "application/x-tar"})
    try:
        with urllib.request.urlopen(req, timeout=60) as r: return json.load(r)["ignored"]
