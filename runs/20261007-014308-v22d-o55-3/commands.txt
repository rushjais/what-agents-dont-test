from probe import q
def cmp(p1, p2, target, extra=()):
    """Both p1,p2 must match target at same level. Returns '>' if p1 beats p2 regardless of order, '<', '=' (last wins), or '?'"""
    files = {target: '', **{e: '' for e in extra}}
    r1 = q({**files, '.packignore': f'{p1}\n!{p2}\n'})  # target ignored => p1 beat
    r2 = q({**files, '.packignore': f'!{p2}\n{p1}\n'})  # target ignored => p1 won or tie
    a = target in r1; b = target in r2
    return {(True, True): '>', (False, False): '<', (False, True): '=', (True, False): '?'}[(a, b)]
EOF

from cmp import cmp
def measure(U, target, name=None):
    name = name or target.rsplit('/',1)[-1]
    n = len(name)
    res = []
    for k in range(n+1):
        cal = name[:k] + '?'*(n-k)   # unanchored, k literal chars
        c = cmp(U, cal, target)
        res.append(c)
        if c == '=':
            return k
        if c == '<':
            return f'<{k}' if k == 0 else f'between {k-1} and {k}: {res}'
    return f'>{n}'
EOF
