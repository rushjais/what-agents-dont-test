    """Return the files under `root` that its .packignore files exclude.

    Only files are listed, never directories. A `.git` directory directly

import fuzz
def mine_of(tree):
    d = tempfile.mkdtemp()
    for p, c in tree.items():
        full = os.path.join(d, p); os.makedirs(os.path.dirname(full), exist_ok=True)
        with open(full, 'w', newline='') as f: f.write(c or '')

def theirs_of(tree):
    t = q(tree)
    return 'ERR' if isinstance(t, tuple) else t
def bad(tree): return mine_of(tree) != theirs_of(tree)
def shrink(tree):
    changed = True
    while changed:
        changed = False
        for k in list(tree):
            t2 = dict(tree); del t2[k]
            if bad(t2): tree = t2; changed = True; continue
            if tree[k] is not None:
                lines = tree[k].splitlines(True)
                for i in range(len(lines)):
                    t2 = dict(tree); t2[k] = ''.join(lines[:i]+lines[i+1:])
