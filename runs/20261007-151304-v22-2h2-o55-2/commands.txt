import os, sys, subprocess, tempfile, json, shutil
def run(files, modes=None):
    """files: dict path->content (str or bytes). Returns ignored list."""
    d = tempfile.mkdtemp(dir="/tmp/p/trees")
    for p, c in files.items():
        full = os.path.join(d, p)
        if p.endswith("/"):
            os.makedirs(full, exist_ok=True); continue
        os.makedirs(os.path.dirname(full), exist_ok=True)
        with open(full, "wb") as f:
            f.write(c.encode() if isinstance(c, str) else c)
        if modes and p in modes: os.chmod(full, modes[p])
    r = subprocess.run([sys.executable, "/work/repo/tools/packignore", d], capture_output=True, text=True)
    shutil.rmtree(d)
    if r.returncode: return "ERR:" + r.stderr.strip()
    return r.stdout.split()

def batch(cases):
    """cases: list of (pi_content, files(list or dict)). Each in tN/. Returns list of sorted ignored lists (relative, excluding tN/.packignore)."""
    d = {}
    for i, (pi, files) in enumerate(cases):
        if pi is not None: d[f't{i}/.packignore'] = pi
        if isinstance(files, dict):
            for k, v in files.items(): d[f't{i}/{k}'] = v
        else:
            for k in files: d[f't{i}/{k}'] = ''
    r = run(d)
    if isinstance(r, str): raise Exception(r)
    out = [[] for _ in cases]
    for p in r:
        t, rest = p.split('/', 1)
        if rest == '.packignore': continue
        out[int(t[1:])].append(rest)
    return out

def scores(items):
    """items: list of (X, path). Returns score per item via ladder on basename (score of basename[:k]+'*' == k)."""
    cases=[]; idx=[]
    for X, path in items:
        base = path.rsplit('/',1)[-1]
        ks=[]
        for k in range(0, len(base)+1):
            L = base[:k] + '*'
            ks.append(len(cases)); cases.append((f'{X}\n!{L}\n', [path]))
            cases.append((f'!{L}\n{X}\n', [path]))
        idx.append(ks)
    res = batch(cases)
    out=[]
    for (X,path), ks in zip(items, idx):
        beat=[bool(res[i]) for i in ks]   # X earlier beats L_k later -> s(X)>k
        later=[bool(res[i+1]) for i in ks] # X later beats L_k earlier -> s(X)>=k
        if not later[0]:
            out.append('NOMATCH'); continue
        s = sum(beat)
        ok = all(beat[k]==(k<s) for k in range(len(beat))) and all(later[k]==(k<=s) for k in range(len(later)))
        out.append(s if ok else ('INCONSISTENT', beat, later))
    return out

def scores2(items):
    cases=[]; idx=[]
    for X, path in items:
        ks=[]
        for k in range(0, 16):
            L = '*{' + 'Q'*k + ',}'
            ks.append(len(cases)); cases.append((f'{X}\n!{L}\n', [path]))
            cases.append((f'!{L}\n{X}\n', [path]))
        idx.append(ks)
    res = batch(cases)
    out=[]
    for (X,path), ks in zip(items, idx):
        beat=[bool(res[i]) for i in ks]
        later=[bool(res[i+1]) for i in ks]
        if not later[0]:
            out.append('NOMATCH'); continue
        s = sum(beat)
        ok = all(beat[k]==(k<s) for k in range(len(beat))) and all(later[k]==(k<=s) for k in range(len(later)))
        out.append(s if ok else ('INCONSISTENT', beat, later))
    return out

    """Split a pattern into literal strings and `{a,b}` groups (lists).

    Each group runs from a `{` to the first `}` after it; groups do not nest
    and an unclosed `{` is literal.

class _GlobMatcher:
    """Match a braced glob without expanding it.

    Equivalent to fully matching the alternation of `_translate_glob` over
    every brace expansion, but the number of expansions can be huge. The
    search walks the pattern pieces directly, carrying the context the
    translation depends on: the run of `*`s so far (capped at 3), and whether
    it follows the start of the pattern or a slash that a `**/` consumed
    (_BOUNDARY), a plain `/` not yet matched (_SLASH), or anything else.
    """

    def __init__(self, pieces, anchored):
        self.strings = [p if isinstance(p, list) else [p] for p in pieces]
        self.anchored = anchored

    def _next(self, k, a, o):
        """Yield the pattern positions (k, a, o) holding the next character."""
        if o < len(self.strings[k][a]):
            yield (k, a, o)
            return
        k += 1
        if k == len(self.strings):
            yield None
            return
        for a2 in range(len(self.strings[k])):
            yield from self._next(k, a2, 0)

    def __call__(self, path):
        n = len(path)
        seen = set()
        stack = [(pos, _BOUNDARY, 0, 0) for pos in self._next(0, 0, 0)] if self.strings[0] else []
        if not stack:

        while stack:

            seen.add(state)

                        return True

                        continue

                stack.extend((p, prev, min(run + 1, 3), i) for p in after)
                continue
            if prev == _SLASH:

                i += 1
            if globstar and c == "/":
