import json, os, random, shutil, subprocess, sys, tempfile
sys.path.insert(0, "/work/repo")
from snapshot.ignore import ignored_files

NAMES = ["a", "b", "ab", "a.b", "x", "d", "e", "ba", ".a", "#a", ";a", "!a", "a b", "[a]", "a\\b", "é", "aa", "bab"]
CHUNKS = ["a", "b", "x", "d", "e", ".", "*", "**", "?", "/", "!", "#", ";", " ", "\\", "[", "]", "é", "***", "a*", "*b"]
SEPS = ["\n", "\n", "\n", "\r\n", "\r", "\x0b", " "]

def rand_tree(rng):
    files = set()
    for _ in range(rng.randint(1, 25)):
        depth = rng.choice([1, 1, 2, 2, 3, 4])
        files.add("/".join(rng.choice(NAMES) for _ in range(depth)))
    # drop file/dir conflicts
    dirs = {f.rsplit("/", 1)[0] for f in files if "/" in f}
    alld = set()
    for d in dirs:
        p = d.split("/")
        for i in range(1, len(p) + 1):
            alld.add("/".join(p[:i]))
    files = {f for f in files if f not in alld}
    return files, sorted(alld)

def rand_pattern(rng):
    s = "".join(rng.choice(CHUNKS) for _ in range(rng.randint(1, 5)))
    if rng.random() < 0.3: s = "!" + s
    if rng.random() < 0.15: s = rng.choice([" ", "\t", "  "]) + s
    if rng.random() < 0.15: s = s + rng.choice([" ", "\t"])
    if rng.random() < 0.05: s = ";" + s
    return s

def rand_ignore(rng, names):
    lines = []
    for _ in range(rng.randint(0, 6)):
        if rng.random() < 0.4 and names:
            # path-ish pattern using real names
            k = rng.randint(1, 3)
            segs = [rng.choice(names + ["*", "**", "?"]) for _ in range(k)]
            s = "/".join(segs)
            if rng.random() < 0.3: s = "/" + s
            if rng.random() < 0.3: s += "/"
            if rng.random() < 0.4: s = "!" + s
            lines.append(s)
        else:
            lines.append(rand_pattern(rng))
    text = ""
    for l in lines:
        text += l + rng.choice(SEPS)
    if lines and rng.random() < 0.2: text = text.rstrip("\n")
    return text

def run(seed):
    rng = random.Random(seed)
    files, dirs = rand_tree(rng)
    root = tempfile.mkdtemp(prefix="fz")
    try:
        for f in files:
            p = os.path.join(root, f)
            os.makedirs(os.path.dirname(p), exist_ok=True)
            open(p, "w").close()
        for d in [""] + dirs:
            if rng.random() < (0.9 if d == "" else 0.35):
                ignore = rand_ignore(rng, NAMES)
                p = os.path.join(root, d, ".packignore")
                with open(p, "w", encoding="utf-8", newline="") as fh:
                    fh.write(ignore)
        r = subprocess.run([sys.executable, "/work/repo/tools/packignore", root], capture_output=True, text=True)
        if r.returncode:
            return None
        theirs = sorted(l for l in r.stdout.splitlines() if l)
        # client prints with print(); paths with line separators would break, skip
        mine = ignored_files(root)
        if mine != theirs:
            dump = {}
            for dp, dn, fn in os.walk(root):
                for n in fn:
                    full = os.path.join(dp, n)
                    rel = os.path.relpath(full, root)
                    dump[rel] = open(full, newline="").read() if n == ".packignore" else None
            return dump, sorted(set(theirs) - set(mine)), sorted(set(mine) - set(theirs))
        return True
    finally:
        shutil.rmtree(root)

if __name__ == "__main__":
    start, n = int(sys.argv[1]), int(sys.argv[2])
    bad = errs = 0
    for s in range(start, start + n):
        res = run(s)
        if res is None: errs += 1; continue
        if res is not True:
            bad += 1
            if bad <= 3:
                print("SEED", s); print(json.dumps(res[0], indent=1, ensure_ascii=False)); print("only theirs:", res[1]); print("only mine:", res[2])
    print(f"done {n}: mismatches={bad} errors={errs}")
EOF
