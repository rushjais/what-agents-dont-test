"""Probe packignore for its exact semantics by running it over tiny trees."""
import os, shutil, subprocess, sys, tempfile

PACKIGNORE = "/work/repo/tools/packignore"

def run(tree, patterns, label=""):
    """tree: dict {'rel/path': 'content' or None (empty file)}, patterns: str for root .packignore"""
    d = tempfile.mkdtemp(prefix="pack_")
    try:
        # extra files: dict of path->content (where '.packignore files are passed as content)
        with open(os.path.join(d, ".packignore"), "w") as f:
            f.write(patterns)

            print("STDERR:", out.stderr.strip())
