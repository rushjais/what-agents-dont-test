import os, sys, json, subprocess, tempfile, shutil
def run(files, modes=None):
    """files: dict path->content. returns list from packignore"""
    d = tempfile.mkdtemp(dir='/tmp/w')
    for p, c in files.items():
        full = os.path.join(d, p)
        os.makedirs(os.path.dirname(full), exist_ok=True)
        with open(full, 'w') as f: f.write(c)
        if modes and p in modes: os.chmod(full, modes[p])
    r = subprocess.run([sys.executable, '/work/repo/tools/packignore', d], capture_output=True, text=True)
    shutil.rmtree(d)
    if r.returncode: return 'ERR:' + r.stderr.strip()
    return r.stdout.split()

from q import run
def cmp(files, target, x, y, extra=None):
    """returns which pattern wins for target"""
    res = []
    for pi, neg in [(x+'\n!'+y+'\n', 'y'), ('!'+y+'\n'+x+'\n','y'), (y+'\n!'+x+'\n','x'), ('!'+x+'\n'+y+'\n','x')]:
        d = {f:'' for f in files}; d['.packignore'] = pi
        if extra: d.update(extra)
        r = run(d)
        excl = target in r
        # winner: if excluded, positive pattern won
        neg_wins = not excl
        res.append(neg)  if neg_wins else res.append('x' if neg=='y' else 'y')
    a = ''.join(res)
    if a == 'yyyy': return f'{y} > {x}'
    if a == 'xxxx': return f'{x} > {y}'
    if a == 'yxxy': return f'{x} ~ {y} (last wins)'
    return f'{x} ? {y}: {a}'
EOF

import os, sys, json, tempfile, shutil, io, tarfile, urllib.request
URL = os.environ["PACKIGNORE_URL"]
def query_dir(root):
    buf = io.BytesIO()
    with tarfile.open(fileobj=buf, mode="w:gz") as tar:
        for dirpath, dirnames, filenames in os.walk(root):
            if os.path.samefile(dirpath, root) and ".git" in dirnames:
                dirnames.remove(".git")
            for name in filenames:
                full = os.path.join(dirpath, name)
                if os.path.islink(full) or not os.path.isfile(full):
                    continue
                tar.add(full, arcname=os.path.relpath(full, root).replace(os.sep, "/"))
    req = urllib.request.Request(URL + "/ignored", data=buf.getvalue(), headers={"Content-Type": "application/x-tar"})
    try:
        with urllib.request.urlopen(req, timeout=60) as resp:
            return json.load(resp)["ignored"]
    except urllib.error.HTTPError as e:
        return 'ERR:' + str(json.load(e).get('error'))
def make(files, modes=None, d=None):
    d = d or tempfile.mkdtemp(dir='/tmp/w')
    for p, c in files.items():
        full = os.path.join(d, p)
        os.makedirs(os.path.dirname(full), exist_ok=True)
        with open(full, 'wb') as f: f.write(c if isinstance(c, bytes) else c.encode())
        if modes and p in modes: os.chmod(full, modes[p])
    return d
def run(files, modes=None):
    d = make(files, modes)
    try: return query_dir(d)
    finally: shutil.rmtree(d)
EOF

def query_tar(entries):
    """entries: list of (name, content, mode)"""
    buf = io.BytesIO()
    with tarfile.open(fileobj=buf, mode="w:gz") as tar:
        for name, c, mode in entries:
            c = c if isinstance(c, bytes) else c.encode()
            ti = tarfile.TarInfo(name); ti.size = len(c); ti.mode = mode
            tar.addfile(ti, io.BytesIO(c))
    req = urllib.request.Request(URL + "/ignored", data=buf.getvalue(), headers={"Content-Type": "application/x-tar"})
    try:
        with urllib.request.urlopen(req, timeout=60) as resp:
            return json.load(resp)["ignored"]
    except urllib.error.HTTPError as e:
        return 'ERR:' + str(json.load(e).get('error'))

    paths = list(files) + [d.rstrip('/') for d in dirs if d]
