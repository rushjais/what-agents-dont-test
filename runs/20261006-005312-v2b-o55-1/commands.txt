from q import *
import itertools, sys
def dom(pats, target):
    files={}
    jobs=[]
    i=0
    for A,B in itertools.permutations(pats,2):
        for order in (0,1):
            k=f"k{i}"; i+=1
            files[f"{k}/.packignore"] = f"{A}\n!{B}" if order==0 else f"!{B}\n{A}"
            files[f"{k}/{target}"]=None
            jobs.append((A,B,order,f"{k}/{target}"))
    ig=set(q(files))
    res={}
    for A,B,order,p in jobs: res[(A,B,order)] = p in ig
    M={}
    for A,B in itertools.permutations(pats,2):
        r1,r2=res[(A,B,0)],res[(A,B,1)]
        M[(A,B)] = {(False,True):'=',(True,True):'A',(False,False):'B',(True,False):'?'}[(r1,r2)]
    return M
if __name__=='__main__':
    pats=sys.argv[2:]; M=dom(pats, sys.argv[1])
    w=max(map(len,pats))
    print(' '*w, ' '.join(f'{j}' for j in range(len(pats))))
    for i,A in enumerate(pats):
        print(A.ljust(w), ' '.join('.' if A==B else M[(A,B)] for B in pats), i)
