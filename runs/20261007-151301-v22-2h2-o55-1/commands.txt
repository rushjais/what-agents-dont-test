import sys
sys.path.insert(0, "/tmp/pi")
from h import run
def cmp(A, B, target, extra=None):
    """A, B patterns both matching target file. Returns '>' if A beats B, '<', '='."""
    files = dict(extra or {}); files[target] = ""
    r1 = target in run({**files, ".packignore": f"{A}\n!{B}\n"})  # B last
    r2 = target in run({**files, ".packignore": f"!{B}\n{A}\n"})  # A last
    if r1 and r2: return ">"
    if not r1 and not r2: return "<"
    if not r1 and r2: return "="
    return "?!"
if __name__ == "__main__":
    target = sys.argv[1]
    for a in sys.argv[2:]:
        A, B = a.split(" VS ")
        print(f"{A!r:20} {cmp(A, B, target)} {B!r}")
