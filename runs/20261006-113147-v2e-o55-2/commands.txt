def q(files):

        with urllib.request.urlopen(req,timeout=60) as r: return json.load(r)["ignored"]

def t(ign, paths, sub=None):

    if sub: f.update(sub)
    return q(f)

import random, sys
sys.path.insert(0,'/work/repo')
from q import q
from snapshot.ignore import _ignored
NAMES=['a','b','ab','a.t','.a','x','{a','a}','[a]','a b','!a','#a',';a','A','a,b','\\a','*','?']
TOK=['a','b','x','.','*','**','?','/','/','{','}',',','!',' ','a','t','[',']',';','\\','**/','/**','{a,b}','\t']
def rtree(rng):
    files=set()
    n=rng.randint(1,14)
    for _ in range(n):
        d=rng.randint(1,4)
        files.add('/'.join(rng.choice(NAMES[:rng.choice([4,10,18])]) for _ in range(d)))
    # remove file/dir conflicts
    fs=sorted(files)
    dirs={f.rsplit('/',1)[0] for f in fs if '/' in f}
    alld=set()
    for d in dirs:
        p=d.split('/')
        for i in range(1,len(p)+1): alld.add('/'.join(p[:i]))
    fs=[f for f in fs if f not in alld]
    dirlist=['']+sorted(alld)
    out={f:None for f in fs}
    for d in dirlist:
        if d=='' or rng.random()<0.4:
            out[(d+'/' if d else '')+'.packignore']=rlines(rng, dirlist)
    return out
def rpat(rng, dirlist):
    if rng.random()<0.3:
        d=rng.choice(dirlist) or 'a'
        segs=d.split('/')
        p='/'.join(s if rng.random()<0.7 else rng.choice(['*','**','?','{a,x}']) for s in segs[rng.randint(0,len(segs)-1):])
    else:
        p=''.join(rng.choice(TOK) for _ in range(rng.randint(1,6)))
    if rng.random()<0.25: p='!'+p
    if rng.random()<0.15: p='/'+p
    if rng.random()<0.15: p=p+'/'
    return p
def rlines(rng, dirlist):
    return rng.choice(['\n','\r\n','\n']).join(rpat(rng,dirlist) for _ in range(rng.randint(0,5)))
seed=int(sys.argv[1]); N=int(sys.argv[2])
bad=0
for i in range(N):
    rng=random.Random(seed*100000+i)
    tree=rtree(rng)
    exp=q(tree)
    if isinstance(exp,tuple):
        print('ERR',exp,tree); continue
    got=_ignored([k for k in tree],{k:v for k,v in tree.items() if v is not None})
    if got!=exp:
        bad+=1
        print('MISMATCH',seed*100000+i); print(' tree',tree); print(' exp',exp); print(' got',got)
        if bad>3: break
print('done bad',bad)
EOF
