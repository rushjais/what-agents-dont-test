import io, json, os, tarfile, urllib.request, sys
URL = os.environ["PACKIGNORE_URL"]
def q(files):
    """files: dict path->content (str). returns list or error string"""
    buf = io.BytesIO()
    with tarfile.open(fileobj=buf, mode="w:gz") as tar:
        for p, c in files.items():
            data = c.encode() if isinstance(c, str) else c
            ti = tarfile.TarInfo(p); ti.size = len(data)
            tar.addfile(ti, io.BytesIO(data))
    req = urllib.request.Request(URL + "/ignored", data=buf.getvalue(), headers={"Content-Type": "application/x-tar"})
    try:
        with urllib.request.urlopen(req, timeout=60) as r:
            return json.load(r)["ignored"]
    except urllib.error.HTTPError as e:
        return "ERR: " + str(json.load(e))
def show(pk, paths, sub=None):
    files = {p: "" for p in paths}
    files[(sub + "/" if sub else "") + ".packignore"] = pk
    r = q(files)
    print(repr(pk), "->", r)
    return r

from q import q
NAMES = ['a','b','ab','ba','a.t','b.t','.h','x','d','e','s','a b','{a}','[a]','!a','a\\','é','a,b']
TOK = ['a','b','d','e','s','x','*','?','**','/','{','}',',','!','.','t',';',' ','\\','[',']','#','é']
def rpat(r):
    if r.random()<0.5:
        n=r.randint(1,6); s=''.join(r.choice(TOK) for _ in range(n))
    else:
        segs=[]
        for _ in range(r.randint(1,3)):
            segs.append(r.choice(['*','**','?','a','d','e','s','x','{a,b}','{d,e}','*.t','a*','?b','{a,d/e}','','.h']))
        s='/'.join(segs)
        if r.random()<0.3: s='/'+s
        if r.random()<0.3: s+='/'
    if r.random()<0.3: s='!'+s
    if r.random()<0.1: s=' '+s+' '
    return s
def gen(r):
    files=set(); dirs=['']
    for _ in range(r.randint(1,25)):
        parent=r.choice(dirs)
        name=r.choice(NAMES)
        p=(parent+'/' if parent else '')+name
        if r.random()<0.4 and p not in files:
            if not any(f==p for f in files): dirs.append(p)
        else:
            if p not in dirs: files.add(p)
    # drop files that collide with dirs
    dirset=set(dirs)
    files={f for f in files if f not in dirset and not any(f.startswith(d+'/') for d in [] )}
    tree={}
    for f in files:
        # ensure no file is prefix of another path
        if any(g.startswith(f+'/') for g in files): continue
        tree[f]=''
    for d in dirs:
        if r.random()<0.5:
            pth=(d+'/' if d else '')+'.packignore'
            if any(g.startswith(pth+'/') for g in tree): continue
            tree[pth]='\n'.join(rpat(r) for _ in range(r.randint(1,5)))+'\n'
    # dirs-with-no-files and packignore under a file-name dir conflicts
    ok={}
    for p,c in tree.items():
        parts=p.split('/')
        if any('/'.join(parts[:i]) in tree for i in range(1,len(parts))): continue
        ok[p]=c
    return ok
def run(seed):
    r=random.Random(seed); tree=gen(r)
    exp=q(tree)
    if isinstance(exp,str): return None
    d=tempfile.mkdtemp()
    try:
        for p,c in tree.items():
            full=os.path.join(d,p); os.makedirs(os.path.dirname(full),exist_ok=True)
            open(full,'w').write(c)
        got=ignored_files(d)
    finally: shutil.rmtree(d)
    if got!=exp:
        return tree,exp,got
start=int(sys.argv[1]); n=int(sys.argv[2])
bad=0
for s in range(start,start+n):

print('done, bad',bad)

import sys
from pri import beats
LET='abcdefghijklmnopqrst'
def score(xpat, prefix):
    """xpat must match prefix (a filename part); measured pattern is xpat+'*' on prefix+LET."""
    f = prefix + LET
    X = xpat + '*'
    lo, hi = 0, len(LET)
    for k in range(0, len(LET)+1):
        L = '*' + LET[len(LET)-k:]
        r = beats(X, L, f)
        if r == '=': return k

def bad(t):
    a=q(t)
    if isinstance(a,str): return False
    return a!=mine(t)
def minimize(t):
    changed=True
    while changed:
        changed=False
        for k in list(t):
            t2=dict(t); del t2[k]
            # drop files under removed? keep simple
            if bad(t2): t=t2; changed=True; break
        if changed: continue
        for k in list(t):

                    t2=dict(t); t2[k]='\n'.join(lines[:i]+lines[i+1:])
                    if bad(t2): t=t2; changed=True; break
            if changed: break
    return t
