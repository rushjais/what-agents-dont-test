import random, os, sys, tempfile, shutil, json
sys.path.insert(0, '/work/repo')
from snapshot.ignore import ignored_files
from q import q

NAMES = ['a','b','c','ab','ba','a.b','b.c','.h','x','y','xy','a b',' a','a ','!a','#a','[a]','a]','a\\b','{a,b}','a,b','*','?','a{','}','.packignore','ä','a*']
TOK = ['a','b','c','x','y','.','*','?','**','/','/','/','!','#','{','}',',','[',']','\\',' ','\t','-','ab','.h','.packignore']
def rand_pat(r):
    n = r.randint(1, 7)
    s = ''.join(r.choice(TOK) for _ in range(n))
    if r.random() < 0.3: s = '!' + s
    if r.random() < 0.15: s = ' ' + s
    if r.random() < 0.15: s = s + r.choice([' ', '\t', '\r', '/'])
    return s
def rand_tree(r, prefix):
    files = {}
    dirs = [prefix]
    taken = set()
    for _ in range(r.randint(3, 14)):
        d = r.choice(dirs)
        name = r.choice(NAMES)
        p = (d + '/' if d else '') + name
        if p in taken or any(t.startswith(p + '/') for t in taken) or any(p.startswith(t + '/') for t in taken if t not in dirs):
            continue
        if r.random() < 0.4 and name != '.packignore':
            dirs.append(p); taken.add(p)
        else:
            files[p] = None; taken.add(p)
    for d in dirs:
        if r.random() < (0.9 if d == prefix else 0.35):
            p = (d + '/' if d else '') + '.packignore'
            if p in taken and files.get(p, 0) is not None and p not in files: continue
            sep = r.choice(['\n', '\n', '\r\n'])
            files[p] = sep.join(rand_pat(r) for _ in range(r.randint(1, 6)))
    return files

def materialize(files):
    d = tempfile.mkdtemp()
    for p, c in files.items():
        full = os.path.join(d, p)
        os.makedirs(os.path.dirname(full), exist_ok=True)
        with open(full, 'w', newline='') as f: f.write(c or '')
    return d

def run(seed, ncases=40):
    r = random.Random(seed)
    files = {}
    for i in range(ncases):
        files.update(rand_tree(r, 'k%d' % i))
    # rebuild dict as the client would (empty dirs vanish, all files present)
    d = materialize(files)
    try:
        mine = ignored_files(d)
        theirs = q({p: c for p, c in files.items()})
    finally:
        shutil.rmtree(d)
    if isinstance(theirs, tuple):
        print('ERR', seed, theirs); return
    if mine != theirs:
        ms, ts = set(mine), set(theirs)
        bad = sorted({p.split('/')[0] for p in ms ^ ts})
        for k in bad[:3]:
            print('=== seed', seed, 'case', k)
            for p, c in sorted(files.items()):
                if p.split('/')[0] == k:
                    print('  ', repr(p), '' if c is None else repr(c))
            print('  only mine:', sorted(p for p in ms - ts if p.startswith(k + '/')))
            print('  only theirs:', sorted(p for p in ts - ms if p.startswith(k + '/')))
        return False
    return True

if __name__ == '__main__':
    a, b = int(sys.argv[1]), int(sys.argv[2])
    ok = 0
    for s in range(a, b):
        if run(s): ok += 1
    print('ok', ok, 'of', b - a)
EOF

from q import q
import sys
def compare(triples):
    """triples: (A, B, path). Returns list of '>' (A wins), '<' (B wins), '=' (later wins), or '?'"""
    files = {}
    for i, (a, b, path) in enumerate(triples):
        for j, txt in enumerate(['!%s\n%s' % (a, b), '!%s\n%s' % (b, a)]):
            d = 'k%d_%d' % (i, j)
            files[d + '/.packignore'] = txt
            files[d + '/' + path] = None
    res = set(q(files))
    out = []
    for i, (a, b, path) in enumerate(triples):
        ig0 = ('k%d_0/' % i + path) in res   # !a then b: ignored => b wins
        ig1 = ('k%d_1/' % i + path) in res   # !b then a: ignored => a wins
        out.append({(False, True): '>', (True, False): '<', (True, True): '=', (False, False): 'NEG'}[(ig0, ig1)])
    return out
if __name__ == '__main__':
    path = sys.argv[1]
    pairs = [tuple(x.split(' ~ ')) for x in sys.argv[2:]]
    for (a, b), r in zip(pairs, compare([(a, b, path) for a, b in pairs])):
        print(f'{a!r:20} {r} {b!r}')
