import random, os, sys, tempfile, shutil, json
sys.path.insert(0, "/work/repo")
from snapshot.ignore import ignored_files
from q import q

NAMES = ["a", "b", "ab", "a.t", "b.t", ".a", "c", "x y", "A", "e"]
PTOK = ["a", "b", "*", "?", "**", "/", ".t", ".", " ", "!", "#", "\\", "[a]", "c", "e", "x y"]

def gen_tree(rng):
    files = {}
    dirs = [()]
    for _ in range(rng.randint(1, 14)):
        parent = rng.choice(dirs)
        name = rng.choice(NAMES)
        p = parent + (name,)
        if rng.random() < 0.4:
            dirs.append(p)
        else:
            files[p] = None
    # resolve conflicts: drop files that are prefixes of dirs etc.
    out = {}
    lower = {}
    for p in list(files) + [d + (".packignore",) for d in dirs if rng.random() < 0.6]:
        out[p] = None
    for d in dirs:
        if d and rng.random() < 0.3:
            out[d + ("f",)] = None
    # remove file/dir collisions and case collisions
    paths = sorted(out, key=len)
    final = {}
    seen_lower = {}
    for p in paths:
        prefixes = {p[:i] for i in range(1, len(p))}
        if any(pp in final for pp in prefixes):
            continue
        if any(f[:len(p)] == p and len(f) > len(p) for f in final):
            continue
        key = "/".join(p).lower()
        clash = False
        for i in range(1, len(p) + 1):
            k = "/".join(p[:i]).lower()
            v = "/".join(p[:i])
            if k in seen_lower and seen_lower[k] != v:
                clash = True
        if clash:
            continue
        for i in range(1, len(p) + 1):
            seen_lower["/".join(p[:i]).lower()] = "/".join(p[:i])
        final[p] = None
    res = {}
    for p in final:
        rel = "/".join(p)
        if p[-1] == ".packignore":
            res[rel] = gen_ignore(rng)
        else:
            res[rel] = None
    return res

def gen_ignore(rng):
    lines = []
    for _ in range(rng.randint(0, 5)):
        n = rng.randint(1, 5)
        s = "".join(rng.choice(PTOK) for _ in range(n))
        if rng.random() < 0.3:
            s = "!" + s
        if rng.random() < 0.1:
            s = " " + s
        lines.append(s)
    sep = rng.choice(["\n", "\n", "\r\n"])
    return sep.join(lines) + (sep if rng.random() < 0.7 else "")

def materialize(files):
    d = tempfile.mkdtemp()
    for rel, content in files.items():
        full = os.path.join(d, rel)
        os.makedirs(os.path.dirname(full), exist_ok=True)
        with open(full, "w", newline="") as f:
            f.write(content or "")
    return d

def main(seed0, n):
    bad = 0
    for s in range(seed0, seed0 + n):
        rng = random.Random(s)
        files = gen_tree(rng)
        d = materialize(files)
        try:
            mine = ignored_files(d)
        finally:
            shutil.rmtree(d)
        theirs = q(files)
        if isinstance(theirs, tuple):
            continue
        if mine != theirs:
            bad += 1
            print("SEED", s, json.dumps(files, indent=0))
            print(" mine  ", mine)
            print(" theirs", theirs)
            if bad > 3:
                break
    print("done", bad)

if __name__ == "__main__":
    main(int(sys.argv[1]), int(sys.argv[2]))

    # Classic glob backtracking over path components.
    memo = {}

        key = (i, j)
