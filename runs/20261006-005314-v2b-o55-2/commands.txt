from q import *
def cmp(a, b, paths, target):
    """returns 'A' if a always wins, 'B' if b always wins, 'later' or 'earlier'."""
    r1 = target in t(f'{a}\n!{b}\n', paths)  # a first
    r2 = target in t(f'!{b}\n{a}\n', paths)
    m1 = 'A' if r1 else 'B'; m2 = 'A' if r2 else 'B'
    if m1 == m2: return m1
    return 'later' if (m1=='B') else 'earlier'

from cmp import *
def score(x, path, extra=()):
    """find k s.t. x ties with '*'+suffix(k) of basename (direct match). Returns k or (lo,hi) window"""
    name = path.split('/')[-1]
    P=[path,*extra]
    res=[]
    lo=None
    for k in range(0,len(name)+1):
        b = '*'+name[len(name)-k:] if k else '*'
        if '/' in path: b = '**/'+b  # keep it matching; might affect score
        r = cmp(x,b,P,path)
        res.append(r)
    return res

    files = {}

    for _ in range(rng.randint(1, 25)):

        name = rng.choice(NAMES)
